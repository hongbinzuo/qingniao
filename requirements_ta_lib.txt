# 1. brew install ta-lib
# 2. pip install TA-Lib

# 1. sudo apt-get install libta-lib0-dev
# 2. pip install TA-Lib
#
# Linux (CentOS/RHEL)安装步骤：
# 1. sudo yum install ta-lib-devel
# 2. pip install TA-Lib
