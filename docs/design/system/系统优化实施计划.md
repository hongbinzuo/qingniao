### 3.1 三级止损系统

```python
def calculate_three_tier_stop_loss(entry_price, signal_type, market_data):
    """
    三级止损系统：
    1. 实时流动性止损（优先级最高）
    2. 技术位止损（Vegas通道、支撑阻力等）
    3. 默认止损（保守设置）
    """
    # 第一优先级：实时流动性止损
    order_book = get_order_book_gateio(limit=50)
    if order_book:
        liquidity_analysis = analyze_liquidity_zones(order_book, market_data['current_price'])
        if liquidity_analysis:
            liquidity_stop, info = suggest_stop_loss_by_liquidity(
                entry_price, signal_type, liquidity_analysis
            )
            if liquidity_stop and is_valid_stop_loss(liquidity_stop, entry_price, signal_type):
                return liquidity_stop, "流动性止损", info
    
    # 第二优先级：技术位止损
    technical_stop = calculate_technical_stop_loss(entry_price, signal_type, market_data)
    if technical_stop and is_valid_stop_loss(technical_stop, entry_price, signal_type):
        return technical_stop, "技术位止损", None
    
    # 第三优先级：默认止损
    default_stop = calculate_default_stop_loss(entry_price, signal_type)
    return default_stop, "默认止损（保守）", None

def is_valid_stop_loss(stop_loss, entry_price, signal_type):
    """验证止损是否合理"""
    if signal_type == 'long':
        distance = entry_price - stop_loss
        distance_pct = distance / entry_price
        # 允许0.8%-3%
        return 0.008 <= distance_pct <= 0.03
    else:
        distance = stop_loss - entry_price
        distance_pct = distance / entry_price
        # 允许0.8%-3%
        return 0.008 <= distance_pct <= 0.03
```

### 3.2 三层确认机制

```python
def confirm_signal_with_three_layers(signal, market_data, klines):
    """
    三层确认机制：
    1. 技术指标确认
    2. 模型确认（价格行为、市场结构）
    3. 裸K确认（K线形态、价格行为）
    """
    confirmations = []
    
    # 第一层：技术指标确认
    if confirm_technical_indicators(signal, market_data):
        confirmations.append("技术指标")
    
    # 第二层：模型确认
    if confirm_price_action_model(signal, klines):
        confirmations.append("价格行为模型")
    if confirm_market_structure_model(signal, klines):
        confirmations.append("市场结构模型")
    
    # 第三层：裸K确认
    naked_kline = analyze_naked_kline(klines)
    if confirm_naked_kline(signal, naked_kline):
        confirmations.append("裸K确认")
    
    # 根据确认数量确定信号强度
    if len(confirmations) >= 3:
        signal['strength'] = 'strong'
        signal['confirmed'] = True
    elif len(confirmations) >= 2:
        signal['strength'] = 'medium'
        signal['confirmed'] = True
    else:
        signal['strength'] = 'weak'
        signal['confirmed'] = False
        signal['needs_confirmation'] = True
        signal['missing_confirmations'] = 3 - len(confirmations)
    
    signal['confirmations'] = confirmations
    return signal
```

### 3.3 动态优先级系统

```python
def calculate_dynamic_priority(base_priority, rule_name, market_context, timeframe):
    """
    动态优先级计算
    """
    priority = base_priority
    
    # 市场环境调整
    if market_context.get('trend') == 'strong':
        if 'breakout' in rule_name:
            priority += 20  # 强趋势：突破规则优先级提升
    elif market_context.get('trend') == 'ranging':
        if 'm_top' in rule_name or 'w_bottom' in rule_name:
            priority += 15  # 震荡：M顶/W底规则优先级提升
    
    # 时间框架调整
    if timeframe == '15m' and 'm_top' in rule_name:
        priority += 10  # 15分钟M顶：De.特别关注
    
    # 市场波动性调整
    if market_context.get('volatility', 0) > 0.02:
        if 'fvg' in rule_name:
            priority += 5  # 高波动：FVG规则优先级提升
    
    return priority
```
